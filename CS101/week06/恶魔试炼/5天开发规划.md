```python
# 玩家属性
player = {
    'x': 300,
    'y': 700,
    'vx': 0,
    'vy': 0,
    'width': 40,
    'height': 40,
    'speed': 5,
    'jump_force': -14,  # 统一跳跃力
    'gravity': 0.6,
}

# 移动逻辑
def update_player():
    # 左右移动
    if keyboard.left or keyboard.a:
        player['x'] -= player['speed']
    if keyboard.right or keyboard.d:
        player['x'] += player['speed']
    
    # 边界限制
    player['x'] = max(0, min(WIDTH - player['width'], player['x']))
    
    # 跳跃
    if (keyboard.space or keyboard.w or keyboard.up) and not player['is_jumping']:
        player['vy'] = player['jump_force']
        player['is_jumping'] = True
    
    # 重力
    player['vy'] += player['gravity']
    player['y'] += player['vy']
```

```python
    # 检测最高平台
    highest_platform = min(platforms, key=lambda p: p.y)

def remove_off_screen_platforms():
```

```python
enemies = []

def spawn_enemies():
    # 随机在某些平台上生成敌人
    for platform in platforms:
        if random.random() < 0.3 and platform.y < 600:  # 30%概率
            enemy = {
                'x': platform.x,
                'y': platform.y - 40,
                'width': 40,
                'height': 40,
                'vx': 2,
                'hp': 30,
                'platform': platform
            }
            enemies.append(enemy)

def update_enemies():
    for enemy in enemies:
        enemy['x'] += enemy['vx']
        
        # 碰到平台边界反转
        if enemy['x'] <= enemy['platform'].left or \
           enemy['x'] + enemy['width'] >= enemy['platform'].right:
            enemy['vx'] *= -1

def check_enemy_collision():
    global game_over
    if not player['invincible']:
        player_rect = Rect(player['x'], player['y'], player['width'], player['height'])
        for enemy in enemies:
            enemy_rect = Rect(enemy['x'], enemy['y'], enemy['width'], enemy['height'])
            if player_rect.colliderect(enemy_rect):
                player['hp'] -= 20
                player['invincible'] = True
                player['invincible_timer'] = 60  # 1秒无敌帧（60帧）
                
                if player['hp'] <= 0:
                    game_over = True
```

```python
particles = []

def create_particles(x, y, color, count=10):
    for _ in range(count):
        particle = {
            'x': x,
            'y': y,
            'vx': random.uniform(-3, 3),
            'vy': random.uniform(-3, 3),
            'life': 30,
            'color': color
        }
        particles.append(particle)

def update_particles():
    for p in particles[:]:
        p['x'] += p['vx']
        p['y'] += p['vy']
        p['life'] -= 1
        if p['life'] <= 0:
            particles.remove(p)

def draw_particles():
    for p in particles:
        screen.draw.filled_circle((p['x'], p['y']), 3, p['color'])
```
